% Tag entries with keywords used for filtering and counting. "final" stops the
% map when the match fails, so only matching entries get the keyword; the
% appendstrict step adds a comma only when the entry already has keywords.
\DeclareSourcemap{
  \maps[datatype=bibtex, overwrite]{
    \map{
      \step[fieldsource=type, match=\regexp{Poster}, final]
      \step[fieldset=keywords, fieldvalue={,}, appendstrict]
      \step[fieldset=keywords, fieldvalue={poster}, append]
    }
    \map{
      \step[fieldsource=journal, match=\regexp{arXiv}, final]
      \step[fieldset=keywords, fieldvalue={,}, appendstrict]
      \step[fieldset=keywords, fieldvalue={preprint}, append]
    }
    \map{
      \step[fieldsource=year, match=\regexp{Under\s+Review}, final]
      \step[fieldset=keywords, fieldvalue={,}, appendstrict]
      \step[fieldset=keywords, fieldvalue={underreview}, append]
    }
  }
}

% --- Automatic publication counts ---
% \countbib{\macro}{filters} counts the entries a \printbibliography with
% those filters would print, typesetting them into a discarded box, and
% stores the result in \macro. Use inside a refsection with \nocite{*}.
\newcounter{bibcount}
\defbibenvironment{countonly}
    {\setbox0=\vbox\bgroup}
    {\egroup}
    {\stepcounter{bibcount}}
\newcommand{\countbib}[2]{%
    \setcounter{bibcount}{0}%
    \printbibliography[heading=none, env=countonly, #2]%
    \xdef#1{\arabic{bibcount}}%
}
% \countof{n}{singular}{plural}
\newcommand{\countof}[3]{#1~\ifnum#1=1 #2\else #3\fi}
